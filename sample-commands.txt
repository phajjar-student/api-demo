# Build the Dockerfile
podman build -t php-api-demo .

# Run the Container:
podman run --detach --name my-api-demo-instance --publish 60080:80 --log-driver=journald php-api-demo
# You must use the same name of the container in both commands

# Stop the Container:
podman stop my-api-demo-instance; podman rm my-api-demo-instance

# Enter a running container:
podman exec -it my-api-demo-instance bash

# Executing SQL against a SQLite database.  This only works inside of the container:
sqlite3 /var/lib/sqlite-data/my-api-app.db 'select * from tbl_quotes order by random() limit 1;'

# Parsing JSON via bash and jq:
echo \{\"message\":\"Success\",\"id\":1,\"text\":\"Diplomacy is the art of saying \'Nice doggie\' until you can find a rock.\",\"author\"\:\"Will Rogers\"\} | jq -r '.id'

# Copy files out of a container:
podman cp my-api-demo-instance:/var/www/html/make-access-token.php html/make-access-token.php 
podman cp my-api-demo-instance:/var/www/html/api/index.php html/api/index.php
